import torch
import torch.nn as nn
import torch.optim as optim
import torch.nn.functional as F

#Politica de la red neuronal
"""
input_size: El tamaño del vector de estado N×(d+1)
action_size: El número de acciones posibles (por ejemplo, 4 valores de potencia de transmisión)
"""



class MyNeuralNetwork(nn.Module):
    def __init__(self, input_size, action_size):
        super(MyNeuralNetwork, self).__init__()
        self.fc1 = nn.Linear(input_size, 256)
        self.relu1 = nn.ReLU()
        self.fc2 = nn.Linear(256, 128)
        self.relu2 = nn.ReLU()
        self.fc3 = nn.Linear(128, action_size)
        self.softmax = nn.Softmax(dim=-1)
        #self.initialize_weights()

    def initialize_weights(self):
        for m in self.modules():
            if isinstance(m, nn.Linear):
                nn.init.xavier_uniform_(m.weight)
                nn.init.zeros_(m.bias)

    def forward(self, x):
        #print("First")
        #print(x)

        x1 = self.relu1(self.fc1(x))
        if torch.isnan(x1).any():
            print("NaN detected after fc1")
        x2 = self.relu2(self.fc2(x1))
        if torch.isnan(x2).any():
            print("NaN detected after fc2")
        #print(x2)
        x3 = self.fc3(x2)
        #print("FC3:")
        #print(x3)
        if torch.isnan(x3).any():
            print("NaN detected after fc3")
        #print(x3)
        x_final = self.softmax(x3)
        #print(x_final)
        return x_final


class MyAgent:
    def __init__(self, input_size, action_size, learning_rate=1e-3):
        self.policy_network = MyNeuralNetwork(input_size, action_size)
        self.optimizer = optim.Adam(self.policy_network.parameters(), lr=learning_rate)
        self.log_probs = []
        self.rewards = []

    def select_action(self, state):
        state = torch.FloatTensor(state).unsqueeze(0)
        #print("State: ", state)
        probs = self.policy_network(state)
        #print("Probs:", probs)
        m = torch.distributions.Categorical(probs)
        action = m.sample()
        self.log_probs.append(m.log_prob(action))
        return action.item()

    def finish_episode(self):
        total_reward = 0
        returns = []
        for i_reward in self.rewards[::-1]:
            total_reward = i_reward + total_reward
            returns.insert(0, total_reward)
        # Convierte returns a tensor de punto flotante
        returns = torch.tensor(returns, dtype=torch.float32)
        # Verifica si hay más de un retorno y si la desviación estándar es mayor que cero
        if len(returns) > 1 and returns.std(unbiased=False) > 0:
            returns = (returns - returns.mean()) / (returns.std(unbiased=False) + 1e-9)
        else:
            # Si solo hay un dato o la desviación estándar es cero, solo resta la media
            returns = returns - returns.mean()

        loss = []
        for log_prob, total_reward in zip(self.log_probs, returns):
            loss.append(-log_prob * total_reward)
        loss = sum(loss)

        self.optimizer.zero_grad()
        loss.backward()

        torch.nn.utils.clip_grad_norm_(self.policy_network.parameters(), max_norm=1.0)
        self.optimizer.step()

        self.log_probs = []
        self.rewards = []
